      Как подключить в свой проект

      Скопируйте папку models к себе, импортируйте нужную функцию и добавьте результат в сцену. У анимированных моделей в userData.update лежит функция, которую нужно вызывать каждый кадр.

      Те же модели в формате glTF лежат в репозитории в папке threejs-models/glb. Их открывает Blender, а в коде загружает GLTFLoader.

      import * as THREE from 'three';
import { createRobot } from './models/robot.js';

const robot = createRobot();
scene.add(robot);

const clock = new THREE.Clock();
renderer.setAnimationLoop(() => {
  robot.userData.update?.(clock.getElapsedTime());
  renderer.render(scene, camera);
});
